    **Step 3: Create shared auth helper**

    Create `src/app/api/assistant/_lib/auth.ts`:
    ```typescript
    import { NextResponse } from 'next/server';

    export function validateAssistantKey(request: Request): boolean {
      const key = request.headers.get('X-Assistant-Key');
      if (!key || !process.env.ASSISTANT_API_KEY) return false;
      return key === process.env.ASSISTANT_API_KEY;
    }

    export function unauthorizedResponse() {
      return NextResponse.json(
        { error: 'Unauthorized' },
        { status: 401 }
      );
    }
    ```

    Create `src/app/api/assistant/_lib/serialize.ts`. Base this on the existing pattern in `src/app/(dashboard)/loans/utils.ts` but make it more general:
    ```typescript
    export function serializeDecimal(value: any): number | null {
      if (value === null || value === undefined) return null;
      if (typeof value === 'number') return value;
      if (typeof value.toNumber === 'function') return value.toNumber();
      return Number(value);
    }

    export function serializeLoan(loan: any) {
      return {
        ...loan,
        loanAmount: serializeDecimal(loan.loanAmount),
        purchasePrice: serializeDecimal(loan.purchasePrice),
        ltv: serializeDecimal(loan.ltv),
        rate: serializeDecimal(loan.rate),
        compPercent: serializeDecimal(loan.compPercent),
        estimatedCompAmount: serializeDecimal(loan.estimatedCompAmount),
        actualCompAmount: serializeDecimal(loan.actualCompAmount),
      };
    }

    export function serializeLead(lead: any) {
      return {
        ...lead,
        leadCost: serializeDecimal(lead.leadCost),
        loanAmountRequested: serializeDecimal(lead.loanAmountRequested),
        propertyValue: serializeDecimal(lead.propertyValue),
        currentMortgageBalance: serializeDecimal(lead.currentMortgageBalance),
        cashOutAmount: serializeDecimal(lead.cashOutAmount),
        proposedLtv: serializeDecimal(lead.proposedLtv),
        currentLtv: serializeDecimal(lead.currentLtv),
      };
    }

    export function serializePartner(partner: any) {
      return {
        ...partner,
        totalVolume: serializeDecimal(partner.totalVolume),
      };
    }
    ```
